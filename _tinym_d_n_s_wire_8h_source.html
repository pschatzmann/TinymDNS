86 static size_t encodeName(const char *name, uint8_t *out, size_t cap) {
87 size_t pos = 0;
88 const char *p = name;
89 while (*p) {
90 const char *dot = strchr(p, '.');
91 size_t len = dot ? (size_t)(dot - p) : strlen(p);
92 if (len == 0 || len > 63) return 0;
93 if (pos + 1 + len + 1 > cap) return 0; // +1 for root terminator
94 out[pos++] = (uint8_t)len;
95 memcpy(out + pos, p, len);
96 pos += len;
97 if (!dot) break;
98 p = dot + 1;
99 }
100 if (pos + 1 > cap || pos + 1 > 255) return 0;
101 out[pos++] = 0;
102 return pos;
103 }

109 static size_t decodeName(const uint8_t *msg, size_t len, size_t pos, char *out, size_t outCap) {
110 size_t next = 0;
111 bool jumped = false;
112 size_t outLen = 0;
113 int hops = 0;
114 if (outCap == 0) return 0;
115 while (true) {
116 if (pos >= len) return 0;
117 uint8_t l = msg[pos];
118 if ((l & 0xC0) == 0xC0) { // compression pointer
119 if (pos + 1 >= len) return 0;
120 if (!jumped) next = pos + 2;
121 jumped = true;
122 if (++hops > 32) return 0; // guard against pointer loops
123 pos = (size_t)(((l & 0x3F) << 8) | msg[pos + 1]);
124 continue;
125 }
126 if (l & 0xC0) return 0; // reserved label types
127 if (l == 0) {
128 if (!jumped) next = pos + 1;
129 break;
130 }
131 if (pos + 1 + l > len) return 0;
132 if (outLen + l + 2 > outCap) return 0;
133 if (outLen) out[outLen++] = '.';
134 memcpy(out + outLen, msg + pos + 1, l);
135 outLen += l;
136 pos += 1 + l;
137 }
138 out[outLen] = 0;
139 return next;
140 }

143 static size_t buildQuery(uint8_t *buf, size_t cap, const char *fqdn, uint16_t type) {
144 if (cap < kHeaderSize) return 0;
145 memset(buf, 0, kHeaderSize);
146 writeU16(buf + 4, 1); // QDCOUNT
147 size_t n = encodeName(fqdn, buf + kHeaderSize, cap - kHeaderSize);
148 if (!n) return 0;
149 size_t pos = kHeaderSize + n;
150 if (pos + 4 > cap) return 0;
151 writeU16(buf + pos, type);
152 writeU16(buf + pos + 2, kClassIN);
153 return pos + 4;
154 }

162 static size_t buildAResponse(uint8_t *buf, size_t cap, uint16_t id, const char *fqdn,
163 IPAddress ip, uint32_t ttl, bool legacy) {
164 if (cap < kHeaderSize) return 0;
165 memset(buf, 0, kHeaderSize);
166 writeU16(buf, id);
167 writeU16(buf + 2, kFlagResponse);
168 writeU16(buf + 4, legacy ? 1 : 0); // QDCOUNT
169 writeU16(buf + 6, 1); // ANCOUNT
170 size_t pos = kHeaderSize;
171
172 size_t n;
173 if (legacy) {
174 n = encodeName(fqdn, buf + pos, cap - pos);
175 if (!n || pos + n + 4 > cap) return 0;
176 pos += n;
177 writeU16(buf + pos, kTypeA);
178 writeU16(buf + pos + 2, kClassIN);
179 pos += 4;
180 }
181
182 n = encodeName(fqdn, buf + pos, cap - pos);
183 if (!n || pos + n + 10 + 4 > cap) return 0;
184 pos += n;
185 writeU16(buf + pos, kTypeA);
186 writeU16(buf + pos + 2, legacy ? kClassIN : (kClassIN | kUnicastFlag)); // cache-flush
187 if (legacy && ttl > 10) ttl = 10;
188 writeU32(buf + pos + 4, ttl);
189 writeU16(buf + pos + 8, 4); // RDLENGTH
190 pos += 10;
191 for (int i = 0; i < 4; i++) buf[pos + i] = ip[i];
192 pos += 4;
193 return pos;
194 }

200 static bool parseAResponse(const uint8_t *msg, size_t len, const char *fqdn,
201 IPAddress &ip, uint32_t &ttl) {
202 if (len < kHeaderSize) return false;
203 if (!(readU16(msg + 2) & 0x8000)) return false; // not a response
204 uint16_t qd = readU16(msg + 4);
205 uint16_t an = readU16(msg + 6);
206 uint16_t ns = readU16(msg + 8);
207 uint16_t ar = readU16(msg + 10);
208
209 char name[TINYMDNS_NAME_MAX];
210 size_t pos = kHeaderSize;
211 for (uint16_t i = 0; i < qd; i++) {
212 size_t next = decodeName(msg, len, pos, name, sizeof(name));
213 if (!next || next + 4 > len) return false;
214 pos = next + 4;
215 }
216 uint32_t records = (uint32_t)an + ns + ar;
217 for (uint32_t i = 0; i < records; i++) {
218 size_t next = decodeName(msg, len, pos, name, sizeof(name));
219 if (!next || next + 10 > len) return false;
220 pos = next;
221 uint16_t type = readU16(msg + pos);
222 uint16_t cls = readU16(msg + pos + 2);
223 uint32_t recTtl = readU32(msg + pos + 4);
224 uint16_t rdlen = readU16(msg + pos + 8);
225 pos += 10;
226 if (pos + rdlen > len) return false;
227 if (type == kTypeA && rdlen == 4 && (cls & ~kUnicastFlag) == kClassIN &&
228 equalsIgnoreCase(name, fqdn)) {
229 ip = IPAddress(msg[pos], msg[pos + 1], msg[pos + 2], msg[pos + 3]);
230 ttl = recTtl;
231 return true;
232 }
233 pos += rdlen;
234 }
235 return false;
236 }

238 static void writeU16(uint8_t *p, uint16_t v) {
239 p[0] = (uint8_t)(v >> 8);
240 p[1] = (uint8_t)(v & 0xFF);
241 }
242 static void writeU32(uint8_t *p, uint32_t v) {
243 p[0] = (uint8_t)(v >> 24);
244 p[1] = (uint8_t)(v >> 16);
245 p[2] = (uint8_t)(v >> 8);
246 p[3] = (uint8_t)(v & 0xFF);
247 }
248 static uint16_t readU16(const uint8_t *p) { return (uint16_t)((p[0] << 8) | p[1]); }
249 static uint32_t readU32(const uint8_t *p) {
250 return ((uint32_t)p[0] << 24) | ((uint32_t)p[1] << 16) | ((uint32_t)p[2] << 8) | p[3];
