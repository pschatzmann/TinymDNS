91 bool begin(UDP &udp, const char *hostname, IPAddress address, uint32_t ttl = TINYMDNS_DEFAULT_TTL) {
92 if (!TinymDNSWire::makeLocalName(hostname, fqdn_, sizeof(fqdn_))) return false;
93 udp_ = &udp;
94 address_ = address;
95 ttl_ = ttl;
96#if defined(ESP8266)
97 // ESP8266 multicast join is only on WiFiUDP and needs the interface address first
98 bool joined = static_cast<WiFiUDP *>(udp_)->beginMulticast(IPAddress(0, 0, 0, 0), TINYMDNS_MULTICAST_IP, TINYMDNS_PORT);
99#else
100 bool joined = udp_->beginMulticast(TINYMDNS_MULTICAST_IP, TINYMDNS_PORT);
101#endif
102 multicast_ = joined;
103 if (!joined) {
104 // Fallback for stacks without a working multicast join (e.g. cores built
105 // without IGMP): open a plain UDP socket on the mDNS port. Sending to the
106 // group address still works; receiving queries depends on the stack.
107 if (!udp_->begin(TINYMDNS_PORT)) {
108 started_ = false;
109 return false;
110 }
111 }
112 started_ = true;
113 announce();
114 return true;
115 }

178 bool resolve(const char *hostname, IPAddress &result) {
179 if (!started_ || !hostname) return fail(MDNSResult::NotConfigured);
180
181 if (captive_) {
182 result = address_;
183 lastTtl_ = ttl_;
184 lastResult_ = MDNSResult::Ok;
185 return true;
186 }
187
188 char name[TINYMDNS_NAME_MAX];
189 if (!TinymDNSWire::makeLocalName(hostname, name, sizeof(name))) return fail(MDNSResult::InvalidName);
190
191 uint8_t buf[TINYMDNS_MAX_PACKET_SIZE];
192 size_t n = TinymDNSWire::buildQuery(buf, sizeof(buf), name, TinymDNSWire::kTypeA);
193 if (!n) return fail(MDNSResult::InvalidName);
194
195 strncpy(queryName_, name, sizeof(queryName_) - 1);
196 queryName_[sizeof(queryName_) - 1] = 0;
197 queryAnswered_ = false;
198 queryActive_ = true;
199
200 bool sent = udp_->beginPacket(TINYMDNS_MULTICAST_IP, TINYMDNS_PORT) &&
201 udp_->write(buf, n) == n && udp_->endPacket();
202 if (!sent) {
203 queryActive_ = false;
204 return fail(MDNSResult::SendFailed);
205 }
206
207 uint32_t start = millis();
208 while (millis() - start < timeoutMs_) {
209 poll();
210 if (queryAnswered_) {
211 queryActive_ = false;
212 result = answerIp_;
213 lastTtl_ = answerTtl_;
214 lastResult_ = MDNSResult::Ok;
215 return true;
216 }
217 delay(1);
218 }
219 queryActive_ = false;
220 return fail(MDNSResult::Timeout);
221 }

269 void handlePacket(size_t sz) {
270 IPAddress from = udp_->remoteIP();
271 uint16_t fromPort = udp_->remotePort();
272
273 uint8_t buf[TINYMDNS_MAX_PACKET_SIZE];
274 size_t want = sz < sizeof(buf) ? sz : sizeof(buf);
275 int got = udp_->read(buf, want);
276 while (udp_->available()) udp_->read(); // discard oversized remainder
277 if (got <= 0) return;
278 size_t n = (size_t)got;
279
280 if (n < TinymDNSWire::kHeaderSize) return;
281 uint16_t flags = TinymDNSWire::readU16(buf + 2);
282
283 if (flags & 0x8000) {
284 if (eventCb_) notifyAnswers(buf, n);
285 // Response: capture the answer to our pending resolve() query.
286 if (queryActive_ && !queryAnswered_) {
287 uint32_t ttl = 0;
288 IPAddress ip;
289 if (TinymDNSWire::parseAResponse(buf, n, queryName_, ip, ttl, address_, subnetMask_)) {
290 queryAnswered_ = true;
291 answerIp_ = ip;
292 answerTtl_ = ttl;
293 }
294 }
295 return;
296 }
297 if (flags & 0x7800) return; // only standard queries (opcode 0)
298
299 // Query: answer if it asks for our registered name.
300 uint16_t id = TinymDNSWire::readU16(buf);
301 uint16_t qd = TinymDNSWire::readU16(buf + 4);
302 size_t pos = TinymDNSWire::kHeaderSize;
303
304 bool wantsAnswer = false;
305 bool unicastRequested = false;
306 char name[TINYMDNS_NAME_MAX];
307 for (uint16_t i = 0; i < qd; i++) {
308 size_t next = TinymDNSWire::decodeName(buf, n, pos, name, sizeof(name));
309 if (!next || next + 4 > n) return; // malformed
310 uint16_t type = TinymDNSWire::readU16(buf + next);
311 uint16_t cls = TinymDNSWire::readU16(buf + next + 2);
312 pos = next + 4;
313
314 if ((cls & ~TinymDNSWire::kUnicastFlag) != TinymDNSWire::kClassIN) continue;
315 if (eventCb_) eventCb_(Event::Query, name, from, eventCtx_);
316 if (!TinymDNSWire::equalsIgnoreCase(name, fqdn_)) continue;
317 if (type != TinymDNSWire::kTypeA && type != TinymDNSWire::kTypeANY) continue;
318 wantsAnswer = true;
319 if (cls & TinymDNSWire::kUnicastFlag) unicastRequested = true;
320 }
321 if (!wantsAnswer) return;
322
323 if (eventCb_) eventCb_(Event::ReplySent, fqdn_, from, eventCtx_);
324 if (fromPort != TINYMDNS_PORT) {
325 // Legacy unicast query (RFC 6762 section 6.7): reply to sender with same ID.
326 sendAnswer(from, fromPort, id, true);
327 } else if (unicastRequested) {
328 sendAnswer(from, TINYMDNS_PORT, 0, false);
329 } else {
330 sendAnswer(TINYMDNS_MULTICAST_IP, TINYMDNS_PORT, 0, false);
331 }
332 }

335 void notifyAnswers(const uint8_t *msg, size_t len) {
336 uint16_t qd = TinymDNSWire::readU16(msg + 4);
337 uint32_t records = (uint32_t)TinymDNSWire::readU16(msg + 6) + TinymDNSWire::readU16(msg + 8) +
338 TinymDNSWire::readU16(msg + 10);
339 char name[TINYMDNS_NAME_MAX];
340 size_t pos = TinymDNSWire::kHeaderSize;
341 for (uint16_t i = 0; i < qd; i++) {
342 size_t next = TinymDNSWire::decodeName(msg, len, pos, name, sizeof(name));
343 if (!next || next + 4 > len) return;
344 pos = next + 4;
345 }
346 for (uint32_t i = 0; i < records; i++) {
347 size_t next = TinymDNSWire::decodeName(msg, len, pos, name, sizeof(name));
348 if (!next || next + 10 > len) return;
349 pos = next;
350 uint16_t type = TinymDNSWire::readU16(msg + pos);
351 uint16_t cls = TinymDNSWire::readU16(msg + pos + 2);
352 uint32_t recTtl = TinymDNSWire::readU32(msg + pos + 4);
353 uint16_t rdlen = TinymDNSWire::readU16(msg + pos + 8);
354 pos += 10;
355 if (pos + rdlen > len) return;
356 if (type == TinymDNSWire::kTypeA && rdlen == 4 && (cls & ~TinymDNSWire::kUnicastFlag) == TinymDNSWire::kClassIN) {
357 Event ev = !recTtl ? Event::Goodbye : Event::Announcement;
358 if (recTtl && queryActive_ && !queryAnswered_ && TinymDNSWire::equalsIgnoreCase(name, queryName_)) {
359 ev = Event::ReplyReceived;
360 }
361 eventCb_(ev, name, IPAddress(msg[pos], msg[pos + 1], msg[pos + 2], msg[pos + 3]), eventCtx_);
362 }
363 pos += rdlen;
364 }
365 }

367 bool sendAnswer(IPAddress to, uint16_t port, uint16_t id, bool legacy) {
368 uint8_t buf[TINYMDNS_MAX_PACKET_SIZE];
369 size_t n = TinymDNSWire::buildAResponse(buf, sizeof(buf), id, fqdn_, address_, ttl_, legacy);
370 if (!n) return false;
371 if (!udp_->beginPacket(to, port)) return false;
372 bool ok = udp_->write(buf, n) == n;
373 return udp_->endPacket() && ok;
374 }
