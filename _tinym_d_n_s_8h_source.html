140 bool resolve(const char *hostname, IPAddress &result) {
141 if (!started_ || !hostname) return fail(MDNSResult::NotConfigured);
142
143 char name[TINYMDNS_NAME_MAX];
144 if (!TinymDNSWire::makeLocalName(hostname, name, sizeof(name))) return fail(MDNSResult::InvalidName);
145
146 uint8_t buf[TINYMDNS_MAX_PACKET_SIZE];
147 size_t n = TinymDNSWire::buildQuery(buf, sizeof(buf), name, TinymDNSWire::kTypeA);
148 if (!n) return fail(MDNSResult::InvalidName);
149
150 strncpy(queryName_, name, sizeof(queryName_) - 1);
151 queryName_[sizeof(queryName_) - 1] = 0;
152 queryAnswered_ = false;
153 queryActive_ = true;
154

156 udp_->write(buf, n) == n && udp_->endPacket();
157 if (!sent) {
158 queryActive_ = false;
159 return fail(MDNSResult::SendFailed);

166 queryActive_ = false;

168 lastTtl_ = answerTtl_;
169 lastResult_ = MDNSResult::Ok;
170 return true;

197 void handlePacket(size_t sz) {
198 IPAddress from = udp_->remoteIP();
199 uint16_t fromPort = udp_->remotePort();
200
201 uint8_t buf[TINYMDNS_MAX_PACKET_SIZE];
202 size_t want = sz < sizeof(buf) ? sz : sizeof(buf);
203 int got = udp_->read(buf, want);
204 while (udp_->available()) udp_->read(); // discard oversized remainder
205 if (got <= 0) return;
206 size_t n = (size_t)got;
207
208 if (n < TinymDNSWire::kHeaderSize) return;
209 uint16_t flags = TinymDNSWire::readU16(buf + 2);
210
211 if (flags & 0x8000) {
212 // Response: capture the answer to our pending resolve() query.
213 if (queryActive_ && !queryAnswered_) {
214 uint32_t ttl = 0;
215 IPAddress ip;
216 if (TinymDNSWire::parseAResponse(buf, n, queryName_, ip, ttl)) {
217 queryAnswered_ = true;
218 answerIp_ = ip;
219 answerTtl_ = ttl;
220 }
221 }
222 return;
223 }
224 if (flags & 0x7800) return; // only standard queries (opcode 0)
225
226 // Query: answer if it asks for our registered name.
227 uint16_t id = TinymDNSWire::readU16(buf);
228 uint16_t qd = TinymDNSWire::readU16(buf + 4);
229 size_t pos = TinymDNSWire::kHeaderSize;
230
231 bool wantsAnswer = false;
232 bool unicastRequested = false;
233 char name[TINYMDNS_NAME_MAX];
234 for (uint16_t i = 0; i < qd; i++) {
235 size_t next = TinymDNSWire::decodeName(buf, n, pos, name, sizeof(name));
236 if (!next || next + 4 > n) return; // malformed
237 uint16_t type = TinymDNSWire::readU16(buf + next);
238 uint16_t cls = TinymDNSWire::readU16(buf + next + 2);
239 pos = next + 4;
240
241 if ((cls & ~TinymDNSWire::kUnicastFlag) != TinymDNSWire::kClassIN) continue;
242 if (!TinymDNSWire::equalsIgnoreCase(name, fqdn_)) continue;
243 if (type != TinymDNSWire::kTypeA && type != TinymDNSWire::kTypeANY) continue;
244 wantsAnswer = true;
245 if (cls & TinymDNSWire::kUnicastFlag) unicastRequested = true;
246 }
247 if (!wantsAnswer) return;
248
249 if (fromPort != TINYMDNS_PORT) {
250 // Legacy unicast query (RFC 6762 section 6.7): reply to sender with same ID.
251 sendAnswer(from, fromPort, id, true);
252 } else if (unicastRequested) {
253 sendAnswer(from, TINYMDNS_PORT, 0, false);
254 } else {
255 sendAnswer(TINYMDNS_MULTICAST_IP, TINYMDNS_PORT, 0, false);
256 }
257 }

259 bool sendAnswer(IPAddress to, uint16_t port, uint16_t id, bool legacy) {
260 uint8_t buf[TINYMDNS_MAX_PACKET_SIZE];
261 size_t n = TinymDNSWire::buildAResponse(buf, sizeof(buf), id, fqdn_, address_, ttl_, legacy);
262 if (!n) return false;
263 if (!udp_->beginPacket(to, port)) return false;
264 bool ok = udp_->write(buf, n) == n;
265 return udp_->endPacket() && ok;
266 }
